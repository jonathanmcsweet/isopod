name      : $name
container : $(ctr_name "$name")
engine    : $(meta_get "$name" engine || true)
status    : $("$ENGINE" inspect -f '{{.State.Status}}' "$(ctr_name "$name")" 2>/dev/null || printf 'missing')
ssh       : ssh $(ctr_name "$name")    (127.0.0.1:$(meta_get "$name" port || true), loopback only)
forwards  : $forwards
host-ports: $hostports
secrets   : $secretnames
egress    : $(box_egress_posture "$name")
workspace : $WORKSPACE
sudo      : $([ "$(meta_get "$name" sudo 2>/dev/null || printf 1)" = 1 ] && printf 'yes (in-box user has passwordless sudo)' || printf 'no  (admin via: isopod root-shell '"$name"')')
color     : $(meta_get "$name" color || true)
created   : $(meta_get "$name" created || true)
built     : $(meta_get "$name" built_version 2>/dev/null || printf 'before provenance was recorded')$(box_is_stale "$name" 2>/dev/null && printf '    STALE — run: isopod upgrade %s' "$name")
config    : $(box_config_file "$name")    (edit, then: isopod reconfigure $name)
key       : $(box_dir "$name")/id_ed25519
